<script>

let customers = [];

function addDebt(){

let name = document.getElementById("name").value;
let vehicle = document.getElementById("vehicle").value;
let fuel = document.getElementById("fuel").value;
let liter = document.getElementById("liter").value;
let price = document.getElementById("price").value;
let plate = document.getElementById("plate").value;

if(name==""  vehicle==""  fuel==""  liter==""  price==""){

alert("تکایە هەموو خانەکان پڕ بکەرەوە");
return;

}

let total = liter * price;

let status = "باش";
let statusClass = "good";

if(total >= 300000){

status = "ڕەشکراوە";
statusClass = "bad";

}else if(total >= 100000){

status = "ئاگاداری";
statusClass = "warning";

}

customers.push({
name,
vehicle,
fuel,
liter,
price,
total,
plate,
status
});

reloadTable();
updateDashboard();

document.getElementById("name").value = "";
document.getElementById("vehicle").value = "";
document.getElementById("fuel").value = "";
document.getElementById("liter").value = "";
document.getElementById("price").value = "";
document.getElementById("plate").value = "";

}

function reloadTable(){

document.getElementById("tableBody").innerHTML = "";

customers.forEach(c => {

let statusClass = "good";

if(c.status=="ئاگاداری"){
statusClass = "warning";
}

if(c.status=="ڕەشکراوە"){
statusClass = "bad";
}

let row = `
<tr>

<td>${c.name}</td>
<td>${c.vehicle}</td>
<td>${c.fuel}</td>
<td>${c.liter} L</td>
<td>${c.price} د.ع</td>
<td>${c.total} د.ع</td>
<td>${c.plate}</td>

<td class="${statusClass}">
${c.status}
</td>

<td>

<button class="small-btn red"
onclick="deleteCustomer('${c.name}')">
سڕینەوە
</button>

</td>

</tr>
`;

document.getElementById("tableBody").innerHTML += row;

});

}

function updateDashboard(){

let totalDebt = 0;
let totalLiter = 0;

customers.forEach(c => {

totalDebt += Number(c.total);
totalLiter += Number(c.liter);

});

document.getElementById("totalDebt").innerText =
totalDebt + " د.ع";

document.getElementById("totalLiter").innerText =
totalLiter + " L";

document.getElementById("customerCount").innerText =
customers.length;

document.getElementById("todayDebt").innerText =
totalDebt + " د.ع";

}

function searchCustomer(){

let search =
document.getElementById("searchInput").value;

let found =
customers.find(c => c.name === search);

if(found){

document.getElementById("searchResult").innerHTML = `

<p><strong>ناو:</strong> ${found.name}</p><br>

<p><strong>جۆری ئۆتۆمبێل:</strong>
${found.vehicle}</p><br>

<p><strong>سووتەمەنی:</strong>
${found.fuel}</p><br>

<p><strong>لیتەر:</strong>
${found.liter} L</p><br>

<p><strong>کۆی قەرز:</strong>
${found.total} د.ع</p><br>

<p><strong>دۆخ:</strong>
${found.status}</p>

`;

}else{

document.getElementById("searchResult").innerHTML =
"هیچ کڕیارێک نەدۆزرایەوە";

}

}

function deleteCustomer(name){

customers =
customers.filter(c => c.name !== name);

reloadTable();
updateDashboard();

}

function refreshSystem(){

if(confirm("دڵنیایت لە ریفریشکردنی سیستەم؟")){

customers = [];

reloadTable();
updateDashboard();

document.getElementById("searchResult").innerHTML =
"هیچ داتایەک نییە";

alert("سیستەم ریفریش کرا");

}

}

function markAsPaid(){

let paidName =
document.getElementById("paidName").value;

customers =
customers.filter(c => c.name !== paidName);

reloadTable();
updateDashboard();
alert("قەرز پاک کرایەوە");

}

function blacklistCustomer(){

let blackName =
document.getElementById("blackName").value;

customers.forEach(c => {

if(c.name === blackName){

c.status = "ڕەشکراوە";

}

});

reloadTable();

alert("کڕیار ڕەش کرایەوە");

}
